ISP# vim /etc/net/ifaces/enp0s3/options
TYPE=eth
BOOTPROTO=dhcp

создаем каталоги для интерфейсов enp0s8 и enp0s9

ISP# mkdir /etc/net/ifaces/enp0s8
ISP# mkdir /etc/net/ifaces/enp0s9

редактируем созданные интерфейсы

ISP# vim /etc/net/ifaces/enp0s8/options
TYPE=eth
BOOTPROTO=static
ISP# vim /etc/net/ifaces/enp0s9/options
TYPE=eth
BOOTPROTO=static
ISP# echo "172.16.4.1/28" > /etc/net/ifaces/enp0s8/ipv4address
ISP# echo "172.16.5.1/28" > /etc/net/ifaces/enp0s9/ipv4address

меняем net.ipv4.ip_forward на 1

ISP# vim /etc/net/sysctl.conf

настраиваем iptables

ISP# iptables -t nat -A POSTROUTING -o enp0s3 -j MASQUERADE
ISP# iptables-save >> /etc/sysconfig/iptables
ISP# systemctl enable --now iptables.service

рестарт

ISP# systemctl restart network

HQ-RTR

задаем имя

HQ-RTR# hostnamectl  set-hostname hq-rtr.au-team.irpo; exec bash

задаем часовой пояс

HQ-RTR# timedatectl set-timezone Europe/Moscow

настраиваем интерфейс enp0s3

HQ-RTR# vim /etc/net/ifaces/enp0s3/options
TYPE=eth
BOOTPROTO=static
HQ-RTR# echo "172.16.4.14/28" >/etc/net/ifaces/enp0s3/ipv4address
HQ-RTR# echo "default via 172.16.4.1" >/etc/net/ifaces/enp0s3/ipv4route
HQ-RTR# echo "nameserver 77.88.8.8" > /etc/net/ifaces/enp0s3/resolv.conf

создаем каталог для enp0s8

HQ-RTR# mkdir /etc/net/ifaces/enp0s8

редактируем

HQ-RTR# vim /etc/net/ifaces/enp0s8/options
TYPE=eth
BOOTPROTO=static
HQ-RTR# echo "192.168.100.1/26" > /etc/net/ifaces/enp0s8/ipv4address

создаем каталоги для vlan

HQ-RTR# mkdir /etc/net/ifaces/vlan100
HQ-RTR# mkdir /etc/net/ifaces/vlan200
HQ-RTR# mkdir /etc/net/ifaces/vlan999

редактируем 3 файла options

HQ-RTR# vim /etc/net/ifaces/vlan100/options
TYPE=ovsport
BRIDGE=HQ-SW
VID=100
BOOTPROTO=static
CONFIG_IPV4=yes
HQ-RTR# cp /etc/net/ifaces/vlan100 /etc/net/ifaces/vlan200/options
HQ-RTR# cp /etc/net/ifaces/vlan100/options /etc/net/ifaces/vlan999/options
HQ-RTR# vim /etc/net/ifaces/vlan200/optqins
VID=200
HQ-RTR# vim /etc/net/ifaces/vlan999/options
VID=999

задаем ip адреса для интерфейсов vlan

HQ-RTR# echo "192.168.100.1/26" > /etc/net/ifaces/vlan100/ipv4address
HQ-RTR# echo "192.168.100.65/28" > /etc/net/ifaces/vlan200/ipv4address
HQ-RTR# echo "192.168.100.81/29" > /etc/net/ifaces/vlan999/ipv4address

создаем интерфейс hq-sw и настраиваем

HQ-RTR# mkdir /etc/net/ifaces/HQ-SW
HQ-RTR# echo "TYPE=ovsbr" > /etc/net/ifaces/HQ-SW/options

хз

HQ-RTR# modprobe 8021q

меняем net.ipv4.ip_forward на 1

HQ-RTR# vim /etc/net/sysctl.conf

хз

HQ-RTR# sed -i "s/OVS_REMOVE=yes/OVS_REMOVE=no/g" /etc/net/ifaces/default/options

создаем пользователя и настраиваем

HQ-RTR# useradd net_admin
HQ-RTR# passwd net_admin
HQ-RTR# usermod -aG wheel net_admin
HQ-RTR# echo "net_admin ALL=(ALL:ALL) NOPASSWD: ALL" >> /etc/sudoers

создаем ip тунель

HQ-RTR# mkdir /etc/net/ifaces/gre1
HQ-RTR# vim /etc/net/ifaces/gre1/options
TYPE=iptun
TUNTYPE=gre
TUNLOCAL=172.16.4.14
TUNREMOTE=172.16.5.14
TUNOPTIONS='ttl 64'
HOST=enp0s3
HQ-RTR# echo "10.10.10.1/30" > /etc/net/ifaces/gre1/ipv4address
HQ-RTR# modprobe gre

настраиваем iptables

HQ-RTR# iptables -t nat -A POSTROUTING -o enp0s3 -j MASQUERADE
HQ-RTR# iptables-save >> /etc/sysconfig/iptables
HQ-RTR# systemctl enable --now iptables.service

рестарт

HQ-RTR# systemctl restart network
